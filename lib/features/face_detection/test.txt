The Double Snackbar Bug — Root Cause
Here's exactly what's happening:
faces > 1 detected
      │
      ▼
state: shouldShowMultiFaceWarning = true   ← ref.listen fires (1st snackbar)
      │
      ▼
resetMultiFaceWarning() called inside listener
      │
      ▼
state: shouldShowMultiFaceWarning = false  ← ref.listen fires AGAIN
      │                                      previous=true, next=false
      ▼                                      condition passes again
2nd snackbar shown ← BUG                    because we check previous==false
The fix: auto-reset inside the notifier using Future.microtask — the UI never needs to call reset manually.

dartimport 'dart:async';
import 'package:camera/camera.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:basic_project_setup/core/services/permission_service.dart';

class FaceCaptureNotifier extends StateNotifier<FaceCaptureState> {
  final MediaPermissionHandler _permissionHandler;
  final FaceDetectorService _faceDetectorService;

  DateTime _lastProcessed = DateTime(0);
  static const _throttleDuration = Duration(milliseconds: 300);

  // ── Multi-face snackbar cooldown ──────────────────────────────
  DateTime _lastMultiFaceWarning = DateTime(0);
  static const _multiFaceCooldown = Duration(seconds: 3);

  // ── Hold timer ────────────────────────────────────────────────
  Timer? _holdTimer;
  double _holdProgress = 0.0;
  static const _holdDuration = Duration(milliseconds: 1500);
  static const _timerTickMs = 50; // tick every 50ms → smooth arc

  CameraDescription? _selectedCamera;

  static const double _leftAlignedMin = -45.0;
  static const double _leftAlignedMax = -25.0;
  static const double _frontAlignedMin = -12.0;
  static const double _frontAlignedMax = 12.0;
  static const double _rightAlignedMin = 25.0;
  static const double _rightAlignedMax = 45.0;

  FaceCaptureNotifier(
    this._permissionHandler,
    this._faceDetectorService,
  ) : super(const FaceCaptureState());

  Future<void> initialize() async {
    if (state.status == CameraStatus.loading) return;
    state = state.copyWith(status: CameraStatus.loading);

    final result = await _permissionHandler.ensureCameraAccess();
    if (result != PermissionResult.granted) {
      state = state.copyWith(status: CameraStatus.permissionDenied);
      return;
    }

    try {
      final cameras = await availableCameras();
      _selectedCamera = cameras.firstWhere(
        (c) => c.lensDirection == CameraLensDirection.front,
        orElse: () => cameras.first,
      );

      final controller = CameraController(
        _selectedCamera!,
        ResolutionPreset.high,
        enableAudio: false,
        imageFormatGroup: ImageFormatGroup.yuv420,
      );

      await controller.initialize();
      await _startImageStream(controller);

      state = state.copyWith(
        status: CameraStatus.ready,
        cameraController: controller,
      );
    } catch (e) {
      state = state.copyWith(
        status: CameraStatus.error,
        errorMessage: e.toString(),
      );
    }
  }

  Future<void> _startImageStream(CameraController controller) async {
    await controller.startImageStream((CameraImage image) {
      final now = DateTime.now();
      if (now.difference(_lastProcessed) < _throttleDuration) return;
      _lastProcessed = now;
      _processFrame(image);
    });
  }

  Future<void> _processFrame(CameraImage image) async {
    if (_selectedCamera == null) return;

    final faces = await _faceDetectorService.detectFaces(
      cameraImage: image,
      camera: _selectedCamera!,
    );

    // ── No face ───────────────────────────────────────────────
    if (faces.isEmpty) {
      _cancelHoldTimer(); // face left frame — reset arc
      state = state.copyWith(
        alignState: FaceAlignState.idle,
        faceCount: 0,
        eulerY: 0.0,
        holdProgress: 0.0,
      );
      return;
    }

    // ── Multiple faces ────────────────────────────────────────
    if (faces.length > 1) {
      _cancelHoldTimer();

      final now = DateTime.now();
      final cooldownExpired =
          now.difference(_lastMultiFaceWarning) > _multiFaceCooldown;

      if (cooldownExpired) {
        _lastMultiFaceWarning = now;
        state = state.copyWith(
          alignState: FaceAlignState.wrong,
          faceCount: faces.length,
          holdProgress: 0.0,
          shouldShowMultiFaceWarning: true,
        );

        // ✅ FIX: auto-reset the flag after this frame completes
        // UI sees true → shows snackbar → next microtask resets to false
        // ref.listen won't fire a second time because we only
        // show snackbar when previous==false AND next==true
        Future.microtask(() {
          if (mounted) {
            state = state.copyWith(shouldShowMultiFaceWarning: false);
          }
        });
      } else {
        state = state.copyWith(
          alignState: FaceAlignState.wrong,
          faceCount: faces.length,
          holdProgress: 0.0,
        );
      }
      return;
    }

    // ── Single face ───────────────────────────────────────────
    final raw = faces.first.headEulerAngleY ?? 0.0;
    final eulerY = raw * -1;
    final aligned = _isAlignedForCurrentStep(eulerY);

    state = state.copyWith(
      alignState: aligned ? FaceAlignState.aligned : FaceAlignState.wrong,
      eulerY: eulerY,
      faceCount: 1,
    );

    if (aligned) {
      _startHoldTimerIfNeeded(); // ← begin countdown
    } else {
      _cancelHoldTimer();        // ← face drifted, reset arc
    }
  }

  // ── Hold timer ────────────────────────────────────────────────

  void _startHoldTimerIfNeeded() {
    // Don't restart if already running
    if (_holdTimer?.isActive ?? false) return;

    _holdProgress = 0.0;

    _holdTimer = Timer.periodic(
      Duration(milliseconds: _timerTickMs),
      (timer) async {
        _holdProgress += _timerTickMs / _holdDuration.inMilliseconds;

        if (_holdProgress >= 1.0) {
          _holdProgress = 1.0;
          timer.cancel();
          await _capturePhoto(); // 🎯 take the shot
        }

        // Push progress to state so arc repaints
        if (mounted) {
          state = state.copyWith(holdProgress: _holdProgress);
        }
      },
    );
  }

  void _cancelHoldTimer() {
    _holdTimer?.cancel();
    _holdTimer = null;
    _holdProgress = 0.0;
  }

  // ── Capture ───────────────────────────────────────────────────

  Future<void> _capturePhoto() async {
    final controller = state.cameraController;
    if (controller == null || !controller.value.isInitialized) return;

    try {
      // Pause the stream so takePicture doesn't conflict
      await controller.stopImageStream();

      final photo = await controller.takePicture();

      final updatedPhotos = [...state.capturedPhotos, photo];

      state = state.copyWith(
        capturedPhotos: updatedPhotos,
        holdProgress: 0.0,
        alignState: FaceAlignState.idle,
      );

      // Resume stream for next step
      await _startImageStream(controller);
    } catch (e) {
      // If capture fails, just reset and let user try again
      _cancelHoldTimer();
      await _startImageStream(state.cameraController!);
    }
  }

  bool _isAlignedForCurrentStep(double eulerY) {
    // Step 7 replaces this with a switch on currentStep
    return eulerY >= _leftAlignedMin && eulerY <= _leftAlignedMax;
  }

  Future<void> disposeCamera() async {
    _cancelHoldTimer();
    await state.cameraController?.stopImageStream();
    await state.cameraController?.dispose();
  }

  @override
  void dispose() {
    _cancelHoldTimer();
    state.cameraController?.stopImageStream();
    state.cameraController?.dispose();
    super.dispose();
  }
}

class FaceCaptureState extends Equatable {
  final CameraStatus status;
  final CameraController? cameraController;
  final String? errorMessage;
  final FaceAlignState alignState;
  final double eulerY;
  final int faceCount;
  final bool shouldShowMultiFaceWarning;

  // ── Step 6 additions ──────────────────────────────────────────
  final double holdProgress;          // 0.0 → 1.0 drives the arc
  final List<XFile> capturedPhotos;   // grows: [], [left], [left,front], [left,front,right]

  const FaceCaptureState({
    this.status = CameraStatus.idle,
    this.cameraController,
    this.errorMessage,
    this.alignState = FaceAlignState.idle,
    this.eulerY = 0.0,
    this.faceCount = 0,
    this.shouldShowMultiFaceWarning = false,
    this.holdProgress = 0.0,
    this.capturedPhotos = const [],
  });

  FaceCaptureState copyWith({
    CameraStatus? status,
    CameraController? cameraController,
    String? errorMessage,
    FaceAlignState? alignState,
    double? eulerY,
    int? faceCount,
    bool? shouldShowMultiFaceWarning,
    double? holdProgress,
    List<XFile>? capturedPhotos,
  }) {
    return FaceCaptureState(
      status: status ?? this.status,
      cameraController: cameraController ?? this.cameraController,
      errorMessage: errorMessage ?? this.errorMessage,
      alignState: alignState ?? this.alignState,
      eulerY: eulerY ?? this.eulerY,
      faceCount: faceCount ?? this.faceCount,
      shouldShowMultiFaceWarning:
          shouldShowMultiFaceWarning ?? this.shouldShowMultiFaceWarning,
      holdProgress: holdProgress ?? this.holdProgress,
      capturedPhotos: capturedPhotos ?? this.capturedPhotos,
    );
  }

  @override
  List<Object?> get props => [
        status,
        errorMessage,
        alignState,
        eulerY,
        faceCount,
        shouldShowMultiFaceWarning,
        holdProgress,
        capturedPhotos.length, // length is enough — avoids deep list compare
      ];
}
